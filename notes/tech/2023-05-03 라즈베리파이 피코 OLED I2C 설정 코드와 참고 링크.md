from utime import sleep  
from machine import Pin, I2C  
import SSD1306  

pin = Pin("LED", Pin.OUT)  

pin.toggle()  
sleep(1)  
pin.toggle()  
sleep(1)  
pin.toggle()  


# I2C設定 (I2C識別ID 0or1, SDA, SCL)  
i2c = I2C(1, sda=Pin(14), scl=Pin(15), freq=400000)  

# 使用するSSD1306のアドレス取得表示（通常は0x3C）  
addr = i2c.scan()  
print( "OLED I2C Address :" + str(addr) )
